package data

import (
	"errors"
	"net/http"
	"time"

	"github.com/jimmitjoo/tjo/auth"
	up "github.com/upper/db/v4"
)

type Token struct {
	ID        int       `db:"id,omitempty" json:"id"`
	UserID    int       `db:"user_id" json:"user_id"`
	FirstName string    `db:"first_name" json:"first_name"`
	Email     string    `db:"email" json:"email"`
	// PlainText is never persisted: db:"-" keeps it out of the tokens table.
	// It is populated only by GenerateToken, so the caller can hand it to the
	// user once. A Token loaded from the database has it empty, which is why
	// serialising one cannot leak a usable bearer token.
	PlainText string `db:"-" json:"token,omitempty"`

	// Hash is what the database stores and what lookups match on.
	Hash []byte `db:"token_hash" json:"-"`
	CreatedAt time.Time `db:"created_at" json:"created_at"`
	UpdatedAt time.Time `db:"updated_at" json:"updated_at"`
	ExpiresAt time.Time `db:"expiry" json:"expires_at"`
}

func (t *Token) Table() string {
	return "tokens"
}

func (t *Token) GetUserForToken(token string) (*User, error) {
	var user User
	var theToken Token

	collection := upper.Collection(t.Table())
	res := collection.Find(up.Cond{"token_hash =": hashToken(token)})
	err := res.One(&theToken)

	if err != nil {
		return nil, err
	}

	collection = upper.Collection(user.Table())
	res = collection.Find(up.Cond{"id =": theToken.UserID})
	err = res.One(&user)

	if err != nil {
		return nil, err
	}

	user.Token = theToken

	return &user, nil
}

// get tokens for a user
func (t *Token) GetTokensForUser(id int) ([]*Token, error) {
	collection := upper.Collection(t.Table())

	var tokens []*Token

	res := collection.Find(up.Cond{"user_id =": id}).OrderBy("created_at desc")
	err := res.All(&tokens)

	if err != nil {
		return nil, err
	}

	return tokens, nil
}

// get a token by id
func (t *Token) Find(id int) (*Token, error) {
	collection := upper.Collection(t.Table())

	var token Token

	res := collection.Find(up.Cond{"id =": id})
	err := res.One(&token)

	if err != nil {
		return nil, err
	}

	return &token, nil
}

// hashToken returns the value stored for a bearer token. Tokens are looked up
// by hash so that read access to the database -- a backup, a dump, a logged
// slow query -- does not hand over usable credentials.
//
// The framework's implementation, not a second one. Two copies of the same
// hashing is how the two stop agreeing, and this one is tested.
func hashToken(token string) []byte {
	return auth.HashToken(token)
}

// get token by token
func (t *Token) GetByToken(token string) (*Token, error) {
	collection := upper.Collection(t.Table())

	var theToken Token

	res := collection.Find(up.Cond{"token_hash =": hashToken(token)})
	err := res.One(&theToken)

	if err != nil {
		return nil, err
	}

	return &theToken, nil
}

// delete token by id
func (t *Token) Delete(id int) error {
	collection := upper.Collection(t.Table())

	err := collection.Find(id).Delete()

	return err
}

// delete token by token
func (t *Token) DeleteByToken(token string) error {
	collection := upper.Collection(t.Table())

	err := collection.Find(up.Cond{"token_hash =": hashToken(token)}).Delete()

	return err
}

// insert token by token and user
func (t *Token) Insert(token Token, user User) error {
	collection := upper.Collection(t.Table())

	// delete user's existing tokens
	err := collection.Find(up.Cond{"user_id =": user.ID}).Delete()

	if err != nil {
		return err
	}

	token.CreatedAt = time.Now()
	token.UpdatedAt = time.Now()
	token.FirstName = user.FirstName
	token.Email = user.Email

	_, err = collection.Insert(token)

	return err
}

// GenerateToken mints a bearer token for a user.
//
// auth.NewToken produces the plaintext and the hash; only the hash is stored,
// and the plaintext exists in the returned value and nowhere else. That is the
// correction v0.7.0 made after tokens were persisted in plaintext and
// serialised into JSON responses -- a logged response body was a working
// credential.
func (t *Token) GenerateToken(userId int, ttl time.Duration) (*Token, error) {
	minted, err := auth.NewToken(ttl)
	if err != nil {
		return nil, err
	}

	return &Token{
		UserID:    userId,
		PlainText: minted.PlainText,
		Hash:      minted.Hash,
		ExpiresAt: minted.Expiry,
		CreatedAt: time.Now(),
		UpdatedAt: time.Now(),
	}, nil
}

// Authenticate a token
func (t *Token) AuthenticateToken(r *http.Request) (*User, error) {
	authirazationHeader := r.Header.Get("Authorization")
	if authirazationHeader == "" {
		return nil, errors.New("no authorization header received")
	}

	// Parsing the header is auth's, including the length check: a value of the
	// wrong length cannot be one of ours and does not need a database lookup to
	// find out.
	token, err := auth.FromAuthorizationHeader(authirazationHeader)
	if err != nil {
		return nil, err
	}

	tkn, err := t.GetByToken(token)

	if err != nil {
		return nil, errors.New("no matching token found")
	}

	if tkn.ExpiresAt.Before(time.Now()) {
		return nil, errors.New("token has expired")
	}

	user, err := t.GetUserForToken(token)

	if err != nil {
		return nil, errors.New("no matching user found")
	}

	return user, nil
}

// ValidateToken reports whether a bearer token is usable.
//
// One error for unknown and expired, because distinguishing them tells an
// attacker which of their guesses existed.
func (t *Token) ValidateToken(token string) (bool, error) {
	if _, err := t.GetUserForToken(token); err != nil {
		return false, auth.ErrInvalidToken
	}

	tkn, err := t.GetByToken(token)
	if err != nil || tkn.ExpiresAt.Before(time.Now()) {
		return false, auth.ErrInvalidToken
	}

	return true, nil
}
