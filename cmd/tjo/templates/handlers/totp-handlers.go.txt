// Two-factor authentication.
//
// Nothing here decides whether a code is valid, how a secret is generated, or
// what a recovery code is worth. auth.VerifyTOTP does that, and it does two
// things the code this replaced did not: it rejects a code that has already
// been used, as RFC 6238 §5.2 requires, and it does its comparisons in constant
// time.
//
// The one thing an OTP library is still used for is turning the provisioning
// URI into a QR image. Drawing a picture is not a security decision.

// TwoFactorSetup shows the QR code and the recovery codes.
func (h *Handlers) TwoFactorSetup(w http.ResponseWriter, r *http.Request) {
	userID := h.App.HTTP.Session.GetInt(r.Context(), "userID")
	if userID == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	user, err := h.Models.Users.Find(userID)
	if err != nil {
		h.App.Logging.Error.Println("error finding user:", err)
		h.App.Error500(w, r)
		return
	}

	if user.TOTPEnabled {
		h.App.HTTP.Session.Put(r.Context(), "error", "Two-factor authentication is already enabled")
		http.Redirect(w, r, "/user/settings", http.StatusSeeOther)
		return
	}

	secret, err := auth.NewTOTPSecret()
	if err != nil {
		h.App.Logging.Error.Println("error generating TOTP secret:", err)
		h.App.Error500(w, r)
		return
	}

	// Held in the session, not in the users table: 2FA is not enabled until a
	// code from the authenticator proves the secret arrived intact. Writing it
	// first would lock the account behind a secret nobody has.
	h.App.HTTP.Session.Put(r.Context(), "totp_setup_secret", secret)

	qrCode, err := h.qrCode(auth.TOTPURI(h.App.AppName, user.Email, secret))
	if err != nil {
		h.App.Logging.Error.Println("error rendering QR code:", err)
		h.App.Error500(w, r)
		return
	}

	vars := make(jet.VarMap)
	vars.Set("qrCode", qrCode)
	vars.Set("secret", secret)

	if err := h.App.HTTP.Render.Page(w, r, "2fa-setup", vars, nil); err != nil {
		h.App.Logging.Error.Println("error rendering 2fa-setup:", err)
	}
}

// qrCode renders a provisioning URI as a base64 PNG for embedding in the page.
func (h *Handlers) qrCode(uri string) (string, error) {
	key, err := otp.NewKeyFromURL(uri)
	if err != nil {
		return "", err
	}

	img, err := key.Image(200, 200)
	if err != nil {
		return "", err
	}

	var buf bytes.Buffer
	if err := png.Encode(&buf, img); err != nil {
		return "", err
	}
	return base64.StdEncoding.EncodeToString(buf.Bytes()), nil
}

// TwoFactorVerifySetup enables 2FA once the authenticator has proved it holds
// the secret, and issues the recovery codes.
func (h *Handlers) TwoFactorVerifySetup(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()

	userID := h.App.HTTP.Session.GetInt(ctx, "userID")
	if userID == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing form:", err)
		h.App.Error500(w, r)
		return
	}

	secret := h.App.HTTP.Session.GetString(ctx, "totp_setup_secret")
	if secret == "" {
		h.App.HTTP.Session.Put(ctx, "error", "Setup session expired. Please try again.")
		http.Redirect(w, r, "/user/2fa/setup", http.StatusSeeOther)
		return
	}

	step, err := auth.VerifyTOTP(secret, r.Form.Get("code"), time.Now(), 0)
	if err != nil {
		h.App.HTTP.Session.Put(ctx, "error", "Invalid verification code. Please try again.")
		http.Redirect(w, r, "/user/2fa/setup", http.StatusSeeOther)
		return
	}

	if err := h.Models.Users.EnableTOTP(userID, secret); err != nil {
		h.App.Logging.Error.Println("error enabling 2FA:", err)
		h.App.Error500(w, r)
		return
	}
	if err := h.Models.Users.SetTOTPLastStep(userID, step); err != nil {
		h.App.Logging.Error.Println("error recording TOTP step:", err)
	}

	// Without these, losing a phone means losing the account, and the recovery
	// procedure is an edit against the database.
	codes, err := auth.NewRecoveryCodes(ctx, h.ResetTokens(), strconv.Itoa(userID), recoveryCodeTTL)
	if err != nil {
		h.App.Logging.Error.Println("error issuing recovery codes:", err)
		h.App.Error500(w, r)
		return
	}

	h.App.HTTP.Session.Remove(ctx, "totp_setup_secret")

	// Shown once. They are stored as hashes, so this page is the only place
	// they exist in readable form.
	vars := make(jet.VarMap)
	vars.Set("codes", codes)

	if err := h.App.HTTP.Render.Page(w, r, "2fa-recovery-codes", vars, nil); err != nil {
		h.App.Logging.Error.Println("error rendering 2fa-recovery-codes:", err)
	}
}

// recoveryCodeTTL is long by nature: a recovery code is useful precisely when
// someone has not been able to log in for a while.
const recoveryCodeTTL = 10 * 365 * 24 * time.Hour

// TwoFactorDisable shows the confirmation page.
func (h *Handlers) TwoFactorDisable(w http.ResponseWriter, r *http.Request) {
	userID := h.App.HTTP.Session.GetInt(r.Context(), "userID")
	if userID == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	user, err := h.Models.Users.Find(userID)
	if err != nil {
		h.App.Logging.Error.Println("error finding user:", err)
		h.App.Error500(w, r)
		return
	}

	if !user.TOTPEnabled {
		h.App.HTTP.Session.Put(r.Context(), "error", "Two-factor authentication is not enabled")
		http.Redirect(w, r, "/user/settings", http.StatusSeeOther)
		return
	}

	if err := h.App.HTTP.Render.Page(w, r, "2fa-disable", nil, nil); err != nil {
		h.App.Logging.Error.Println("error rendering 2fa-disable:", err)
	}
}

// TwoFactorDisablePost turns 2FA off, after proving the person doing it holds
// the authenticator.
func (h *Handlers) TwoFactorDisablePost(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()

	userID := h.App.HTTP.Session.GetInt(ctx, "userID")
	if userID == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing form:", err)
		h.App.Error500(w, r)
		return
	}

	user, err := h.Models.Users.Find(userID)
	if err != nil {
		h.App.Logging.Error.Println("error finding user:", err)
		h.App.Error500(w, r)
		return
	}

	step, err := auth.VerifyTOTP(user.TOTPSecret, r.Form.Get("code"), time.Now(), user.TOTPLastStep)
	if err != nil {
		h.App.HTTP.Session.Put(ctx, "error", "Invalid verification code.")
		http.Redirect(w, r, "/user/2fa/disable", http.StatusSeeOther)
		return
	}
	if err := h.Models.Users.SetTOTPLastStep(userID, step); err != nil {
		h.App.Logging.Error.Println("error recording TOTP step:", err)
	}

	if err := h.Models.Users.DisableTOTP(userID); err != nil {
		h.App.Logging.Error.Println("error disabling 2FA:", err)
		h.App.Error500(w, r)
		return
	}

	// The recovery codes went with it. Leaving them valid would keep a second
	// way past a second factor that is no longer there.
	if err := h.ResetTokens().InvalidateUser(ctx, strconv.Itoa(userID), auth.PurposeRecoveryCode); err != nil {
		h.App.Logging.Error.Println("error invalidating recovery codes:", err)
	}

	h.App.HTTP.Session.Put(ctx, "flash", "Two-factor authentication has been disabled.")
	http.Redirect(w, r, "/user/settings", http.StatusSeeOther)
}

// TwoFactorChallenge shows the code form during login.
func (h *Handlers) TwoFactorChallenge(w http.ResponseWriter, r *http.Request) {
	if h.App.HTTP.Session.GetInt(r.Context(), "pending_2fa_user_id") == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	if err := h.App.HTTP.Render.Page(w, r, "2fa-challenge", nil, nil); err != nil {
		h.App.Logging.Error.Println("error rendering 2fa-challenge:", err)
	}
}

// TwoFactorVerify completes a login that was waiting on a second factor.
//
// A recovery code is accepted in place of an authenticator code, and is spent
// atomically when it is -- one code, one use, enforced by the store rather than
// by a read followed by a delete.
func (h *Handlers) TwoFactorVerify(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()

	pendingUserID := h.App.HTTP.Session.GetInt(ctx, "pending_2fa_user_id")
	if pendingUserID == 0 {
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing form:", err)
		h.App.Error500(w, r)
		return
	}

	user, err := h.Models.Users.Find(pendingUserID)
	if err != nil {
		h.App.Logging.Error.Println("error finding user:", err)
		h.App.Error500(w, r)
		return
	}

	if recovery := r.Form.Get("recovery_code"); recovery != "" {
		owner, err := auth.UseRecoveryCode(ctx, h.ResetTokens(), recovery)
		// The code has to belong to the account that is halfway through this
		// login. Without the comparison, anyone's valid recovery code would
		// finish anyone else's challenge.
		if err != nil || owner != strconv.Itoa(pendingUserID) {
			h.App.HTTP.Session.Put(ctx, "error", "Invalid recovery code.")
			http.Redirect(w, r, "/2fa/challenge", http.StatusSeeOther)
			return
		}
	} else {
		step, err := auth.VerifyTOTP(user.TOTPSecret, r.Form.Get("code"), time.Now(), user.TOTPLastStep)
		if err != nil {
			h.App.HTTP.Session.Put(ctx, "error", "Invalid verification code. Please try again.")
			http.Redirect(w, r, "/2fa/challenge", http.StatusSeeOther)
			return
		}
		// Recorded before the session is granted, so a code cannot be spent
		// twice even by two requests arriving together.
		if err := h.Models.Users.SetTOTPLastStep(user.ID, step); err != nil {
			h.App.Logging.Error.Println("error recording TOTP step:", err)
		}
	}

	remember := h.App.HTTP.Session.GetBool(ctx, "pending_remember")
	h.App.HTTP.Session.Remove(ctx, "pending_2fa_user_id")
	h.App.HTTP.Session.Remove(ctx, "pending_remember")

	// The same function the password-only path ends in, so the session renewal
	// and the CSRF rotation cannot differ between them.
	h.completeLogin(w, r, user, remember)
}
