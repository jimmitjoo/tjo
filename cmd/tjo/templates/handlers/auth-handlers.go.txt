func (h *Handlers) UserRegister(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	err := h.App.HTTP.Render.Page(w, r, "register", nil, nil)

	if err != nil {
		h.App.Logging.Error.Println("error rendering register:", err)
	}
}

func (h *Handlers) PostUserRegister(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	err := r.ParseForm()
	if err != nil {
		h.App.Logging.Error.Println("error parsing register form:", err)
	}

	emailAddress := r.Form.Get("email")
	password := r.Form.Get("password")
	verifyPassword := r.Form.Get("verify_password")
	firstName := r.Form.Get("first_name")
	lastName := r.Form.Get("last_name")

	user := data.User{
		FirstName:      firstName,
		LastName:       lastName,
		Email:          emailAddress,
		Password:       password,
		VerifyPassword: verifyPassword,
	}

	validator := h.App.Validator(nil)
	user.Validate(validator)

	if !validator.Valid() {
		h.App.Logging.Error.Println("error validating register form:", validator.Errors)
		h.App.HTTP.Session.Put(r.Context(), "errors", validator.Errors)

		http.Redirect(w, r, "/register", http.StatusSeeOther)
		return
	}

	_, err = h.Models.Users.Create(user)
	if err != nil {
		h.App.Logging.Error.Println("error creating user:", err)
		w.Write([]byte(err.Error()))
		return
	}

	h.sendActivationLink(user)

	h.App.HTTP.Session.Put(r.Context(), "flash", i18n.From(r.Context()).T("auth.registered"))
	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

func (h *Handlers) sendActivationLink(user data.User) {
	// create activation link
	link := fmt.Sprintf("%s/activate-account?email=%s", h.App.Server.URL, user.Email)
	sign := urlsigner.Signer{
		Secret: []byte(h.App.EncryptionKey),
	}

	// sign the link
	signedLink := sign.GenerateTokenFromString(link)

	var data struct {
		Link string
	}
	data.Link = signedLink

	// send email to user with activation link
	msg := email.Message{
		To:       user.Email,
		Subject:  "Welcome to " + os.Getenv("APP_NAME"),
		Template: "welcome",
		Data:     data,
		From:     os.Getenv("MAIL_FROM_ADDRESS"),
	}

	h.App.Background.Mail.Jobs <- msg
	res := <-h.App.Background.Mail.Results
	if res.Error != nil {
		h.App.Logging.Error.Println("error sending email:", res.Error)
		return
	}
}

func (h *Handlers) UserLogin(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	// The identity-provider buttons, which are the configured ones and no
	// others -- a scaffolded project with no credentials in .env shows a plain
	// password form rather than buttons that lead to a 404.
	vars := make(jet.VarMap)
	vars.Set("SocialProviders", h.SocialProviders())

	err := h.App.HTTP.Render.Page(w, r, "login", vars, nil)

	if err != nil {
		h.App.Logging.Error.Println("error rendering login:", err)
	}
}

// PostUserLogin verifies credentials and starts a session.
//
// It decides nothing about the credentials themselves. auth.AuthenticateAndUpgrade
// does the lookup and the bcrypt comparison unconditionally and in that order,
// so an attempt against an address with no account costs the same as one
// against an address with an account -- and it rehashes at the current cost
// while the plaintext is in hand, which is the only moment that is possible.
func (h *Handlers) PostUserLogin(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing login form:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	ctx := r.Context()

	account, err := auth.AuthenticateAndUpgrade(ctx, data.Accounts{},
		r.Form.Get("email"), r.Form.Get("password"),
		func(ctx context.Context, account auth.Account, newHash []byte) {
			id, convErr := strconv.Atoi(account.AuthID())
			if convErr != nil {
				return
			}
			var user *data.User
			if err := user.SetPasswordHash(id, newHash); err != nil {
				// The user authenticated correctly. A storage problem is not
				// their problem, so this is logged and the login proceeds.
				h.App.Logging.Error.Println("error upgrading password hash:", err)
			}
		})

	switch {
	case errors.Is(err, auth.ErrNotActivated):
		// Reported only after the password verified, so it tells nothing to
		// someone who does not already know it. The previous version checked
		// this *before* the password and mailed an activation link, which
		// answered "does this address have an account here" to anyone asking,
		// and mailed a stranger's inbox on request while it was at it.
		id, convErr := strconv.Atoi(account.AuthID())
		if convErr == nil {
			if user, findErr := h.Models.Users.Find(id); findErr == nil {
				h.sendActivationLink(*user)
			}
		}
		h.App.HTTP.Session.Put(ctx, "flash", i18n.From(ctx).T("auth.not_activated"))
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return

	case err != nil:
		// One message for an unknown address and a wrong password, because
		// they are the same answer to the only question a login form should
		// answer.
		h.App.HTTP.Session.Put(ctx, "error", i18n.From(ctx).T("auth.invalid_credentials"))
		http.Redirect(w, r, "/login", http.StatusSeeOther)
		return
	}

	id, err := strconv.Atoi(account.AuthID())
	if err != nil {
		h.App.Logging.Error.Println("account has an unusable id:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	user, err := h.Models.Users.Find(id)
	if err != nil {
		h.App.Logging.Error.Println("error loading the authenticated user:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	remember := r.Form.Get("remember") == "remember"

	if user.TOTPEnabled {
		// Not logged in yet. The session stays anonymous and carries only the
		// fact that a password was accepted, so a request that skips the
		// challenge is a request from an unauthenticated session.
		h.App.HTTP.Session.Put(ctx, "pending_2fa_user_id", user.ID)
		h.App.HTTP.Session.Put(ctx, "pending_remember", remember)
		http.Redirect(w, r, "/2fa/challenge", http.StatusSeeOther)
		return
	}

	h.completeLogin(w, r, user, remember)
}

// completeLogin is the only place in this application where a session becomes
// authenticated.
//
// One function for the password path and the 2FA path, deliberately. These
// three things have to happen together, and the version with a copy per path is
// how one copy ends up missing one of them -- which is exactly what happened
// here before v0.9.0, when the 2FA path renewed the session but did not rotate
// the CSRF token.
//
//   - RenewToken changes the session id, so an attacker who fixed a session id
//     on the victim's browser before login does not hold an authenticated one
//     after it.
//   - RotateCSRFToken because RenewToken keeps the session's *contents*, and
//     the CSRF token lives there; without it the token minted for the anonymous
//     session survives into the authenticated one.
//   - The remember token is issued last, and only once the session is safe.
func (h *Handlers) completeLogin(w http.ResponseWriter, r *http.Request, user *data.User, remember bool) {
	ctx := r.Context()

	if err := h.App.HTTP.Session.RenewToken(ctx); err != nil {
		h.App.Logging.Error.Println("error renewing session token on login:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}
	tjo.RotateCSRFToken(h.App.HTTP.Session, r)

	if remember {
		// Only the hash reaches the database, the token expires, and it is
		// single-use: the next time this cookie is presented it is exchanged
		// for a new one, so a copy of it stops working. All of that is in
		// auth.Remember; this decides whether the box was ticked.
		token, err := auth.Remember(ctx, h.ResetTokens(), strconv.Itoa(user.ID), auth.DefaultRememberTTL)
		if err != nil {
			h.App.Logging.Error.Println("error issuing remember token:", err)
		} else {
			http.SetCookie(w, auth.RememberCookie(
				h.rememberCookieName(), token, auth.DefaultRememberTTL,
				h.App.HTTP.Session.Cookie.Secure))
		}
	}

	h.App.HTTP.Session.Put(ctx, "userID", user.ID)
	h.App.HTTP.Session.Put(ctx, "auth_time", time.Now().Unix())

	http.Redirect(w, r, "/", http.StatusSeeOther)
}

// rememberCookieName is per application, so two applications on one host do not
// present each other's remember cookies.
func (h *Handlers) rememberCookieName() string {
	return fmt.Sprintf("_%s_remember", h.App.AppName)
}

func (h *Handlers) ActivateUserAccount(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	theUrl := r.RequestURI
	testUrl := fmt.Sprintf("%s%s", h.App.Server.URL, theUrl)

	// validate the url
	signer := urlsigner.Signer{
		Secret: []byte(h.App.EncryptionKey),
	}

	valid := signer.VerifyToken(testUrl)
	if !valid {
		h.App.Logging.Error.Println("error validating url")
		h.App.ErrorUnauthorized(w, r)
		return
	}

	// The signature alone only proves the link was issued by us, not that it
	// is still current. Without this an activation URL leaked from an old
	// mailbox stays valid forever.
	if signer.Expired(testUrl, 60*24) {
		h.App.Logging.Error.Println("activation link has expired")
		h.App.ErrorUnauthorized(w, r)
		return
	}

	email := r.URL.Query().Get("email")

	user, err := h.Models.Users.ByEmail(email)
	if err != nil {
		h.App.ErrorStatus(w, http.StatusNotFound)
		return
	}

	err = user.Activate(*user)
	if err != nil {
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	h.App.HTTP.Session.Put(r.Context(), "flash", i18n.From(r.Context()).T("auth.activated"))
	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

// UserLogout ends the session and this browser's remember token.
//
// This device's token only. auth.Forget would end them all, which is not what
// "log out" means on the machine someone is sitting at -- signing a person out
// of their phone because they closed a laptop tab is a surprise, and a silent
// one.
func (h *Handlers) UserLogout(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()

	if token, err := auth.RememberedToken(r, h.rememberCookieName()); err == nil {
		if err := auth.ForgetOne(ctx, h.ResetTokens(), token); err != nil {
			h.App.Logging.Error.Println("error invalidating remember token:", err)
		}
	}
	http.SetCookie(w, auth.ClearRememberCookie(h.rememberCookieName(), h.App.HTTP.Session.Cookie.Secure))

	h.App.HTTP.Session.Destroy(ctx)
	if err := h.App.HTTP.Session.RenewToken(ctx); err != nil {
		h.App.Logging.Error.Println("error renewing session token on logout:", err)
	}
	tjo.RotateCSRFToken(h.App.HTTP.Session, r)

	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

func (h *Handlers) PostForgot(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing forgot form:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	emailAddress := r.Form.Get("email")

	// Every outcome below is identical from the outside, whether or not the
	// address has an account. The previous version looked the user up and then
	// used user.Email regardless, so an unknown address produced a different
	// response from a known one -- which answers "does this person have an
	// account here" to anyone who asks.
	//
	// The work still happens for a known address; it is only the response that
	// is constant.
	var user *data.User
	found, err := user.ByEmail(emailAddress)
	if err == nil && found != nil {
		h.sendPasswordResetLink(found)
	}

	h.App.HTTP.Session.Put(r.Context(), "flash", i18n.From(r.Context()).T("auth.reset_sent"))
	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

// sendPasswordResetLink mints a single-use token and mails it.
//
// The token carries no identity. GHSA-44g2-5v2v-xh66 was a reset flow where the
// artifact presented to the server *was* the identity, encrypted with malleable
// AES-CFB, so flipping bits in it changed whose password was reset. The token is
// now an unguessable lookup key; the identity lives in the row its hash points
// at, and it is consumed atomically when spent.
func (h *Handlers) sendPasswordResetLink(user *data.User) {
	ctx := context.Background()

	token, err := auth.NewResetToken(
		strconv.Itoa(user.ID), auth.PurposePasswordReset, 1*time.Hour)
	if err != nil {
		h.App.Logging.Error.Println("error creating reset token:", err)
		return
	}

	if err := h.ResetTokens().Save(ctx, token); err != nil {
		h.App.Logging.Error.Println("error saving reset token:", err)
		return
	}

	var body struct{ Link string }
	body.Link = fmt.Sprintf("%s/reset-password?token=%s", h.App.Server.URL, token.PlainText)

	msg := email.Message{
		To:       user.Email,
		Subject:  "Reset Password",
		Template: "password-reset",
		Data:     body,
		From:     os.Getenv("MAIL_FROM_ADDRESS"),
	}

	h.App.Background.Mail.Jobs <- msg
	if res := <-h.App.Background.Mail.Results; res.Error != nil {
		h.App.Logging.Error.Println("error sending email:", res.Error)
	}
}

var (
	resetStore     *auth.SQLResetStore
	resetStoreOnce sync.Once
)

// ResetTokens returns the reset-token store, preparing its table on first use.
//
// The implementation is the framework's rather than this project's, because
// Consume has to be atomic and that is exactly what an application writing it
// by hand gets wrong: a SELECT to check the token followed by an UPDATE to mark
// it used lets two concurrent requests both see it as unused, and the prize for
// winning that race is somebody else's account.
func (h *Handlers) ResetTokens() *auth.SQLResetStore {
	resetStoreOnce.Do(func() {
		resetStore = auth.NewSQLResetStore(h.App.Data.DB.Pool, resetDialect(h.App.Config.Database.Type))
		if err := resetStore.Migrate(context.Background()); err != nil {
			h.App.Logging.Error.Println("error preparing the reset token table:", err)
		}
	})
	return resetStore
}

func resetDialect(dbType string) auth.Dialect {
	switch dbType {
	case "postgres", "postgresql", "pgx":
		return auth.DialectPostgres
	case "mysql", "mariadb":
		return auth.DialectMySQL
	default:
		return auth.DialectSQLite
	}
}

// ResetPasswordForm renders the form. It deliberately does not validate the
// token: doing so would tell someone holding a guess whether it was right,
// before they had to commit to using it, and the token is single-use so probing
// it here would also burn it.
func (h *Handlers) ResetPasswordForm(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	vars := make(jet.VarMap)
	vars.Set("token", r.URL.Query().Get("token"))

	if err := h.App.HTTP.Render.Page(w, r, "reset-password", vars, nil); err != nil {
		h.App.Logging.Error.Println("error rendering:", err)
	}
}

// PostResetPassword changes the password.
//
// Everything that decides whether this is allowed lives in the framework's auth
// package: the token is consumed atomically, checked for purpose and expiry,
// the new password is validated against the policy, and every other outstanding
// reset token for the user is invalidated. This handler parses a form and
// writes a row.
//
// The previous version of this function verified nothing at all. See
// GHSA-44g2-5v2v-xh66.
func (h *Handlers) PostResetPassword(w http.ResponseWriter, r *http.Request) {

	h.isAuthenticated(w, r)

	if err := r.ParseForm(); err != nil {
		h.App.Logging.Error.Println("error parsing reset form:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	ctx := r.Context()

	userID, hash, err := auth.ResetPassword(
		ctx, h.ResetTokens(), auth.DefaultPasswordPolicy(),
		r.Form.Get("token"), r.Form.Get("password"))
	if err != nil {
		// One message for an invalid token and for a password that fails the
		// policy. Separating them would say whether the token was good.
		h.App.HTTP.Session.Put(ctx, "error", i18n.From(ctx).T("auth.reset_invalid"))
		http.Redirect(w, r, "/forgot-password", http.StatusSeeOther)
		return
	}

	id, err := strconv.Atoi(userID)
	if err != nil {
		h.App.Logging.Error.Println("reset token has an unusable user id:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	var user *data.User
	if err := user.SetPasswordHash(id, hash); err != nil {
		h.App.Logging.Error.Println("error storing the new password:", err)
		h.App.ErrorStatus(w, http.StatusInternalServerError)
		return
	}

	h.App.HTTP.Session.Put(ctx, "flash", i18n.From(ctx).T("auth.reset_done"))
	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

func (h *Handlers) isAuthenticated(w http.ResponseWriter, r *http.Request) {
	if h.App.HTTP.Session.Exists(r.Context(), "userID") {
		http.Redirect(w, r, "/", http.StatusSeeOther)
		return
	}
}
